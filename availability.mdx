export const LiveCatalogue = () => {
  const [catalogue, setCatalogue] = useState(null);
  const [failed, setFailed] = useState(false);
  useEffect(() => {
    let active = true;
    let busy = false;
    let controller;
    const refresh = async () => {
      if (busy || document.hidden) return;
      busy = true;
      controller = new AbortController();
      try {
        const response = await fetch('https://oddswire.io/api/catalogue', {
          credentials: 'omit', cache: 'no-store',
          signal: AbortSignal.any([controller.signal, AbortSignal.timeout(6000)])
        });
        if (!response.ok) throw new Error('Unavailable');
        const value = await response.json();
        if (value.document?.schema !== 'oddswire-catalogue.v1' || !Array.isArray(value.document.bookmakers) || !Array.isArray(value.document.sports) || value.document.bookmakers.length > 128 || value.document.sports.length > 64) throw new Error('Unavailable');
        if (active) { setCatalogue(value); setFailed(false); }
      } catch { if (active) { setCatalogue(null); setFailed(true); } }
      finally { busy = false; }
    };
    refresh();
    const timer = setInterval(refresh, 15000);
    document.addEventListener('visibilitychange', refresh);
    return () => { active = false; clearInterval(timer); controller?.abort(); document.removeEventListener('visibilitychange', refresh); };
  }, []);
  if (!catalogue) return <p role="status">{failed ? 'Current catalogue and pricing are unavailable. Follow the live directory links below.' : 'Loading the current catalogue and draft prices…'}</p>;
  const config = catalogue.document;
  const money = (value) => new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(value / 100);
  const books = config.bookmakers.filter((book) => book.active);
  const sports = config.sports.filter((sport) => sport.active);
  return <div>
    <p>{books.length} bookmakers and {sports.length} sports. Monthly draft estimates in USD; availability depends on your account coverage.</p>
    <table><thead><tr><th>Bookmaker</th><th>Region package</th><th>Monthly base estimate</th></tr></thead>
      <tbody>{books.map((book) => <tr key={String(book.id)}><th scope="row"><a href={'https://oddswire.io/bookmakers/' + book.collectorId}>{book.name}</a></th><td>{book.region}</td><td>{money(book.priceCents ?? config.pricing.bookmakerCents)}</td></tr>)}</tbody>
    </table>
    <table><thead><tr><th>Sport</th><th>Monthly base estimate</th></tr></thead>
      <tbody>{sports.map((sport) => <tr key={sport.id}><th scope="row"><a href={'https://oddswire.io/sports/' + sport.id}>{sport.name}</a></th><td>{money(sport.priceCents ?? config.pricing.sportCents)}</td></tr>)}</tbody>
    </table>
    {Array.isArray(config.addons) && <table><thead><tr><th>Add-on</th><th>Monthly estimate</th></tr></thead>
      <tbody>{config.addons.filter((addon) => addon.active).map((addon) => <tr key={addon.id}><th scope="row">{addon.name}</th><td>{money(addon.priceCents ?? config.pricing.addonCents)}</td></tr>)}</tbody>
    </table>}
    <p>Live option ×{config.pricing.liveMilli / 1000}; player props ×{config.pricing.propsMilli / 1000}. Selected request tiers multiply the combined bookmaker and sport subtotal. Optional feature fees are added separately. Saving an estimate does not activate access.</p>
  </div>;
};

## Before relying on a feed

Run a free [entitlements check](/reference/getEntitlements). Choose only returned scopes, preserve dataset and source identity, and inspect every `missing[]` entry. A source outage or expired permission can withdraw prices independently of your selected plan.

Current data is intended for bounded pre-match research and explicit queries. There is no published production latency, continuous coverage or high-frequency trading commitment. MCP is for explicit questions rather than automatic price polling.

A free data allowance is not open for general activation. For access, historical availability or permitted display/redistribution, [contact support](mailto:admin@oddswire.io) with your requirements. Do not send credentials.
