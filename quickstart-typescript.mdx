description: "Use Node.js fetch with the hosted API, without installing an unpublished SDK."
---

Use Node.js 22.23.4+ with your key in the private `ODDSWIRE_API_KEY` environment variable. The example uses built-in `fetch`; there is no npm package to install. This runnable example is JavaScript. For a TypeScript integration, add request/response types from the [OpenAPI contract](/openapi.json) and keep requests on the server.

Save it as `check-access.mjs` and run `node check-access.mjs`. Do not put the key in a browser bundle or frontend environment variable.

```javascript
const baseUrl = "https://api.oddswire.io";
const key = process.env.ODDSWIRE_API_KEY;
if (!key || /[\r\n]/.test(key)) throw new Error("Load your own private API key");

async function getPage(path, params = {}) {
  const url = new URL(path, baseUrl);
  for (const [name, value] of Object.entries(params)) url.searchParams.set(name, value);
  const response = await fetch(url, {
    headers: { "X-API-Key": key },
    redirect: "error",
    signal: AbortSignal.timeout(10_000),
  });
  if (!response.ok) {
    const problem = await response.json().catch(() => ({}));
    console.error("Request failed:", response.status, problem.code, problem.request_id);
    throw new Error("Review access and delivery status before retrying");
  }
  return response.json();
}

try {
  const page = await getPage("/v1/account/entitlements"); // Free.
  const access = page.data[0] ?? {};
  console.log("Namespaces:", access.namespaces ?? []);
  console.log("Features:", access.features ?? []);
  const namespace = process.env.ODDSWIRE_NAMESPACE;
  if (namespace) {
    if (!access.namespaces?.includes(namespace)) throw new Error("Namespace not granted");
    if (!access.features?.includes("fixtures")) throw new Error("Fixture access not granted");
    const fixtures = await getPage("/v1/fixtures", { namespace, limit: "1" });
    console.log("Fixture coverage:", fixtures.missing);
    if (fixtures.data.length && access.features.includes("odds")) {
      const fixtureId = String(fixtures.data[0].fixture.id);
      const odds = await getPage("/v1/odds", { namespace, fixture: fixtureId, limit: "3" });
      console.log("Odds rows:", odds.data.length);
      console.log("Price coverage:", odds.missing);
    }
  } else {
    console.log("No data request made. Set an approved namespace only when ready.");
  }
} catch {
  console.error("Request stopped. Check configuration or usage; no automatic retry occurred.");
  process.exitCode = 1;
}
```

With only the key set, this performs one free access check. Explicitly setting an entitled namespace enables up to two data calls, which can consume two credits even for empty pages. An exception does not prove a timed-out delivery was free. Check [Usage](https://oddswire.io/app/usage) before retrying.

Decimal values stay strings. Do not convert prices or sequence IDs to JavaScript `Number`. See [Data and coverage](/data), [Errors and credits](/errors) and the exact [List odds parameters](/reference/listOdds).
