export const LiveCatalogue = () => {
  const [catalogue, setCatalogue] = useState(null);
  const [failed, setFailed] = useState(false);
  useEffect(() => {
    let active = true;
    let busy = false;
    let controller;
    const refresh = async () => {
      if (busy || document.hidden) return;
      busy = true;
      controller = new AbortController();
      try {
        const response = await fetch('https://oddswire.io/api/catalogue', {
          credentials: 'omit', cache: 'no-store',
          signal: AbortSignal.any([controller.signal, AbortSignal.timeout(6000)])
        });
        if (!response.ok) throw new Error('Unavailable');
        const value = await response.json();
        if (value.document?.schema !== 'oddswire-catalogue.v1' || !Array.isArray(value.document.bookmakers) || !Array.isArray(value.document.sports) || value.document.bookmakers.length > 128 || value.document.sports.length > 64) throw new Error('Unavailable');
        if (active) { setCatalogue(value); setFailed(false); }
      } catch { if (active) { setCatalogue(null); setFailed(true); } }
      finally { busy = false; }
    };
    refresh();
    const timer = setInterval(refresh, 15000);
    document.addEventListener('visibilitychange', refresh);
    return () => { active = false; clearInterval(timer); controller?.abort(); document.removeEventListener('visibilitychange', refresh); };
  }, []);
  if (!catalogue) return <p role="status">{failed ? 'Current catalogue and pricing are unavailable. Follow the live directory links below.' : 'Loading the current catalogue and draft prices…'}</p>;
  const config = catalogue.document;
  const money = (value) => new Intl.NumberFormat('en-US', { style: 'currency', currency: 'USD' }).format(value / 100);
  const books = config.bookmakers.filter((book) => book.active);
  const sports = config.sports.filter((sport) => sport.active);
  return <div>
    <p>{books.length} bookmakers and {sports.length} sports. Monthly draft estimates in USD; source coverage and customer permission are separate.</p>
    <table><thead><tr><th>Bookmaker</th><th>Region package</th><th>Monthly base estimate</th></tr></thead>
      <tbody>{books.map((book) => <tr key={String(book.id)}><th scope="row"><a href={'https://oddswire.io/bookmakers/' + book.collectorId}>{book.name}</a></th><td>{book.region}</td><td>{money(book.priceCents ?? config.pricing.bookmakerCents)}</td></tr>)}</tbody>
    </table>
    <table><thead><tr><th>Sport</th><th>Monthly base estimate</th></tr></thead>
      <tbody>{sports.map((sport) => <tr key={sport.id}><th scope="row"><a href={'https://oddswire.io/sports/' + sport.id}>{sport.name}</a></th><td>{money(sport.priceCents ?? config.pricing.sportCents)}</td></tr>)}</tbody>
    </table>
    <p>Live ×{config.pricing.liveMilli / 1000}; player props ×{config.pricing.propsMilli / 1000}. Selected request tiers multiply the combined bookmaker and sport subtotal. Checkout and paid activation remain separate.</p>
  </div>;
};

## Deployed today

| Surface | What is working | What a new user receives |
| --- | --- | --- |
| Public docs and calculators | Hosted documentation and free browser tools | No account is needed for docs or [calculators](https://oddswire.io/tools) |
| Customer workspace | Production email/password login, email verification and automatic personal workspace setup | An account with no API/Terminal data grants or credit allocation |
| Account controls | Keys, usage, coverage and account permissions in the signed-in workspace | Controls appropriate to your role; they do not unlock a feed |
| HTTPS REST API | `https://api.oddswire.io/v1/...` authenticates requests and enforces permissions, freshness and quota | Data requires an explicitly approved scope and allocation |

Production Google/GitHub social sign-in is not configured. Use the email/password flow currently offered by Oddswire; logging in to another service with Google does not configure Oddswire social login.

The REST contract includes both canonical sports data and separately gated native fixture/odds routes. Native reads include source-labelled player-prop candidates; those labels do not establish canonical player identity or settlement compatibility. Kambi distinguishes original retained snapshots from separately approved live mode. Use the [endpoint guide](/endpoints) to choose the correct route family.

Internal source reviews are expiring and founder-only. **They are not customer availability, a data licence, or an entitlement included with signup.** Collection and source controls can change independently of these docs. Read actual account coverage and response freshness; no coverage or uptime guarantee is implied.

| Python/TypeScript SDKs and local data MCP package | Private preview artifacts; no npm/PyPI release or public package installation |
| Billing and beta administration | Restricted test/operator contracts; no public paid checkout, automatic data grant or general beta enrollment |

## Choose your next step

- New account: review [authentication](/authentication), then run a free [access check](/quickstart-curl).
- Approved REST integration: use namespaces and features returned by [Get entitlements](/reference/getEntitlements), then discover fixtures before requesting odds.
- No approved data scope: use the public calculators, inspect the contracts, or contact support. Repeating a denied request will not enable access.

These docs do not change source permissions, country/region restrictions, account expiry, or the applicable use review. Published schemas, successful technical collection and account signup are separate from permission to display or redistribute data.
